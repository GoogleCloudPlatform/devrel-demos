### Example 1: Weekend Trip Planner

User: Plan a weekend trip to Tokyo

Assistant:
Here is a weekend itinerary for Tokyo with highlights for Saturday and Sunday!

```json
{
  "version": "v0.9",
  "createSurface": {
    "surfaceId": "tokyo-weekend-planner",
    "catalogId": "{{CATALOG_ID}}"
  }
}
```

### Example 2: Interactive Reservation Form with Data Binding

User: Book a dinner table in Tokyo

Assistant:
Choose your party size and preferred seating time below to confirm your reservation.

```json
{
  "version": "v0.9",
  "createSurface": {
    "surfaceId": "tokyo-dinner-booking",
    "catalogId": "{{CATALOG_ID}}",
    "sendDataModel": true
  }
}
```

```json
{
  "version": "v0.9",
  "updateComponents": {
    "surfaceId": "tokyo-dinner-booking",
    "components": [
      {
        "id": "root",
        "component": "Card",
        "child": "form-col"
      },
      {
        "id": "form-col",
        "component": "Column",
        "children": [
          "heading",
          "guest-summary",
          "party-slider",
          "time-picker",
          "reserve-btn"
        ],
        "align": "stretch"
      },
      {
        "id": "heading",
        "component": "Text",
        "text": "Dinner Reservation",
        "variant": "h3"
      },
      {
        "id": "guest-summary",
        "component": "Text",
        "text": {
          "call": "formatString",
          "args": {
            "value": "Table for ${/partySize} guests"
          },
          "returnType": "string"
        },
        "variant": "body"
      },
      {
        "id": "party-slider",
        "component": "Slider",
        "label": "Party size",
        "value": {
          "path": "/partySize"
        },
        "min": 1,
        "max": 10
      },
      {
        "id": "time-picker",
        "component": "ChoicePicker",
        "label": "Seating time",
        "value": {
          "path": "/time"
        },
        "variant": "mutuallyExclusive",
        "options": [
          {
            "label": "18:00",
            "value": "18:00"
          },
          {
            "label": "19:30",
            "value": "19:30"
          },
          {
            "label": "21:00",
            "value": "21:00"
          }
        ]
      },
      {
        "id": "reserve-btn-label",
        "component": "Text",
        "text": "Confirm Reservation"
      },
      {
        "id": "reserve-btn",
        "component": "Button",
        "child": "reserve-btn-label",
        "variant": "primary",
        "action": {
          "event": {
            "name": "confirmReservation",
            "context": {
              "partySize": {
                "path": "/partySize"
              },
              "time": {
                "path": "/time"
              }
            }
          }
        }
      }
    ]
  }
}
```

```json
{
  "version": "v0.9",
  "updateDataModel": {
    "surfaceId": "tokyo-dinner-booking",
    "value": {
      "partySize": 4,
      "time": [
        "19:30"
      ]
    }
  }
}
```
